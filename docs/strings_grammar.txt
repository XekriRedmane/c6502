-- Reference documentation. The lexer's terminals are in
-- cpp/generated/c99_lalr.cpp, and cpp/src/lark_lexer.h runs them.

string-literal:
    '"' s-char-sequence? '"'
    | 'L"' s-char-sequence? '"'
    ;

s-char-sequence:
    s-char+ ;

s-char:
    <any member of the source character set except
     the double-quote ("), backslash (\), or new-line character>
    | escape-sequence
    ;

escape-sequence:
    <the same as in chars_grammar.txt>
    ;
